# Créer une carte de la CI
from cartograpy import data, mapper
bound = data.GeoBoundaries()
civ = bound.adm("CIV", "ADM1")
m = mapper.Map(title="Carte simple de la Côte d’Ivoire")
m.add_layer(civ, color="lightyellow", edgecolor="black", linewidth=0.8)
m.add_north_arrow()
m.add_scale_bar()
m.show()
# ✨ Votre carte est prête !
